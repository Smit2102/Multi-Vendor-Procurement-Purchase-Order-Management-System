1. Project Overview

This project implements a comprehensive, role-based enterprise procurement management system. It models the complete lifecycle of organizational purchasing — from an employee submitting a purchase request, through managerial approval, procurement officer purchase order issuance, vendor fulfillment, shipment tracking, invoice generation, and financial payment processing.

The underlying relational database follows a rigorously normalized design satisfying Third Normal Form (3NF). The schema spans 16 interconnected tables organized across six functional modules: User Identity & Access Control, Purchase Request Workflow, Vendor Management, Order Fulfillment, Finance & Payments, and System Monitoring.

The application is built using Python Flask with MySQL as the database system for this assignment. Firebase Authentication handles user login on the client side, while Flask-Login manages server-side session persistence. The system supports six distinct user roles: Employee, Manager, Procurement Officer, Vendor, Finance, and Super Admin — each with dedicated dashboard views, KPI cards, and Chart.js visualizations.

2. Database Schema — MySQL Build Script

The following SQL script creates the database and all 16 tables using MySQL 8.0 syntax. All tables use the InnoDB storage engine for foreign key support. The script is designed to execute sequentially without errors in MySQL Workbench or any MySQL client. 

-- ============================================================
-- DATABASE CREATION
-- ============================================================
CREATE DATABASE IF NOT EXISTS procurement_system;
USE procurement_system;

2.1 Module 1: User Identity & Access Control

-- ============================================================
-- MODULE 1: USER IDENTITY & ACCESS CONTROL
-- ============================================================

-- Table: Roles
-- Purpose: Defines all system roles (Employee, Manager, etc.)
CREATE TABLE Roles (
    role_id          INT AUTO_INCREMENT PRIMARY KEY,
    role_name        VARCHAR(50) UNIQUE NOT NULL,
    permissions_json TEXT
) ENGINE=InnoDB;

-- Table: Departments
-- Purpose: Organizational departments with budget tracking.
CREATE TABLE Departments (
    dept_id          INT AUTO_INCREMENT PRIMARY KEY,
    dept_name        VARCHAR(100) NOT NULL,
    budget_allocated DECIMAL(15,2) DEFAULT 0.00,
    budget_used      DECIMAL(15,2) DEFAULT 0.00,
    manager_id       INT,
    FOREIGN KEY (manager_id) REFERENCES Users(user_id)
        ON DELETE SET NULL
) ENGINE=InnoDB;

-- Table: Users
-- Purpose: All system users with role and department assignment.
CREATE TABLE Users (
    user_id       INT AUTO_INCREMENT PRIMARY KEY,
    name          VARCHAR(100) NOT NULL,
    email         VARCHAR(150) UNIQUE NOT NULL,
    role          INT,
    department_id INT,
    created_at    DATETIME DEFAULT CURRENT_TIMESTAMP,
    is_active     BOOLEAN DEFAULT TRUE,
    FOREIGN KEY (role) REFERENCES Roles(role_id),
    FOREIGN KEY (department_id) REFERENCES Departments(dept_id)
) ENGINE=InnoDB;

2.2 Module 2: Purchase Request Workflow

-- ============================================================
-- MODULE 2: PURCHASE REQUEST WORKFLOW
-- ============================================================

-- Table: Purchase_Requests
-- Purpose: Employee item requests with status tracking.
CREATE TABLE Purchase_Requests (
    pr_id          INT AUTO_INCREMENT PRIMARY KEY,
    employee_id    INT,
    dept_id        INT,
    item_name      VARCHAR(200) NOT NULL,
    description    TEXT,
    quantity       INT NOT NULL,
    estimated_cost DECIMAL(12,2),
    status         VARCHAR(20) DEFAULT 'PENDING',
    created_at     DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (employee_id) REFERENCES Users(user_id),
    FOREIGN KEY (dept_id) REFERENCES Departments(dept_id)
) ENGINE=InnoDB;

-- Table: PR_Approvals
-- Purpose: Manager review decisions with timestamped comments.
CREATE TABLE PR_Approvals (
    approval_id INT AUTO_INCREMENT PRIMARY KEY,
    pr_id       INT,
    manager_id  INT,
    decision    VARCHAR(20),
    comments    TEXT,
    decided_at  DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (pr_id) REFERENCES Purchase_Requests(pr_id),
    FOREIGN KEY (manager_id) REFERENCES Users(user_id)
) ENGINE=InnoDB;

-- Table: PR_Status_History
-- Purpose: Full audit trail of every status change.
CREATE TABLE PR_Status_History (
    history_id INT AUTO_INCREMENT PRIMARY KEY,
    pr_id      INT,
    old_status VARCHAR(20),
    new_status VARCHAR(20),
    changed_by INT,
    changed_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (pr_id) REFERENCES Purchase_Requests(pr_id),
    FOREIGN KEY (changed_by) REFERENCES Users(user_id)
) ENGINE=InnoDB;

2.3 Module 3: Vendor Management

-- ============================================================
-- MODULE 3: VENDOR MANAGEMENT
-- ============================================================

-- Table: Vendors
-- Purpose: External supplier companies with ratings.
CREATE TABLE Vendors (
    vendor_id    INT AUTO_INCREMENT PRIMARY KEY,
    company_name VARCHAR(150) NOT NULL,
    contact_name VARCHAR(100),
    email        VARCHAR(150),
    phone        VARCHAR(30),
    address      VARCHAR(255),
    rating       INT,
    is_active    BOOLEAN DEFAULT TRUE
) ENGINE=InnoDB;

-- Table: Vendor_Categories
-- Purpose: Vendor supply category associations (1:N).
CREATE TABLE Vendor_Categories (
    vc_id         INT AUTO_INCREMENT PRIMARY KEY,
    vendor_id     INT,
    category_name VARCHAR(100),
    FOREIGN KEY (vendor_id) REFERENCES Vendors(vendor_id)
) ENGINE=InnoDB;

-- Table: Contracts
-- Purpose: Formal vendor agreements with terms and ceiling.
CREATE TABLE Contracts (
    contract_id   INT AUTO_INCREMENT PRIMARY KEY,
    vendor_id     INT,
    start_date    DATE,
    end_date      DATE,
    payment_terms VARCHAR(50),
    max_value     DECIMAL(15,2),
    status        VARCHAR(20),
    FOREIGN KEY (vendor_id) REFERENCES Vendors(vendor_id)
) ENGINE=InnoDB;

2.4 Module 4: Order Fulfillment

-- ============================================================
-- MODULE 4: ORDER FULFILLMENT
-- ============================================================

-- Table: Purchase_Orders
-- Purpose: Official PO documents issued to vendors.
CREATE TABLE Purchase_Orders (
    po_id                  INT AUTO_INCREMENT PRIMARY KEY,
    pr_id                  INT,
    vendor_id              INT,
    procurement_officer_id INT,
    issue_date             DATETIME DEFAULT CURRENT_TIMESTAMP,
    delivery_due_date      DATE,
    status                 VARCHAR(20) DEFAULT 'ISSUED',
    total_amount           DECIMAL(15,2),
    FOREIGN KEY (pr_id) REFERENCES Purchase_Requests(pr_id),
    FOREIGN KEY (vendor_id) REFERENCES Vendors(vendor_id),
    FOREIGN KEY (procurement_officer_id) REFERENCES Users(user_id)
) ENGINE=InnoDB;

-- Table: PO_Line_Items
-- Purpose: Individual line items within a purchase order.
CREATE TABLE PO_Line_Items (
    line_id          INT AUTO_INCREMENT PRIMARY KEY,
    po_id            INT,
    item_description VARCHAR(200),
    quantity         INT,
    unit_price       DECIMAL(12,2),
    total_price      DECIMAL(12,2),
    FOREIGN KEY (po_id) REFERENCES Purchase_Orders(po_id)
) ENGINE=InnoDB;

-- Table: Goods_Receipt
-- Purpose: Delivery confirmation records.
CREATE TABLE Goods_Receipt (
    gr_id             INT AUTO_INCREMENT PRIMARY KEY,
    po_id             INT,
    received_by       INT,
    received_date     DATETIME DEFAULT CURRENT_TIMESTAMP,
    quantity_received INT,
    condition_notes   TEXT,
    FOREIGN KEY (po_id) REFERENCES Purchase_Orders(po_id),
    FOREIGN KEY (received_by) REFERENCES Users(user_id)
) ENGINE=InnoDB;

2.5 Module 5: Finance & Payments

-- ============================================================
-- MODULE 5: FINANCE & PAYMENTS
-- ============================================================

-- Table: Invoices
-- Purpose: Billing documents for shipped purchase orders.
CREATE TABLE Invoices (
    invoice_id   INT AUTO_INCREMENT PRIMARY KEY,
    po_id        INT,
    vendor_id    INT,
    invoice_date DATETIME DEFAULT CURRENT_TIMESTAMP,
    due_date     DATE,
    amount       DECIMAL(15,2),
    status       VARCHAR(20) DEFAULT 'PENDING',
    FOREIGN KEY (po_id) REFERENCES Purchase_Orders(po_id),
    FOREIGN KEY (vendor_id) REFERENCES Vendors(vendor_id)
) ENGINE=InnoDB;

-- Table: Payments
-- Purpose: Payment disbursement records against invoices.
CREATE TABLE Payments (
    payment_id     INT AUTO_INCREMENT PRIMARY KEY,
    invoice_id     INT,
    paid_by        INT,
    payment_date   DATETIME DEFAULT CURRENT_TIMESTAMP,
    amount_paid    DECIMAL(15,2),
    payment_method VARCHAR(30),
    reference_no   VARCHAR(50),
    FOREIGN KEY (invoice_id) REFERENCES Invoices(invoice_id),
    FOREIGN KEY (paid_by) REFERENCES Users(user_id)
) ENGINE=InnoDB;

-- Table: Budget_Transactions
-- Purpose: Department budget ledger entries.
CREATE TABLE Budget_Transactions (
    txn_id           INT AUTO_INCREMENT PRIMARY KEY,
    dept_id          INT,
    po_id            INT,
    amount           DECIMAL(15,2),
    transaction_type VARCHAR(20),
    recorded_at      DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (dept_id) REFERENCES Departments(dept_id),
    FOREIGN KEY (po_id) REFERENCES Purchase_Orders(po_id)
) ENGINE=InnoDB;

2.6 Module 6: System & Monitoring

-- ============================================================
-- MODULE 6: SYSTEM & MONITORING
-- ============================================================

-- Table: Audit_Log
-- Purpose: Comprehensive action logging with IP and timestamp.
CREATE TABLE Audit_Log (
    log_id         INT AUTO_INCREMENT PRIMARY KEY,
    user_id        INT,
    action         VARCHAR(50),
    table_affected VARCHAR(50),
    timestamp      DATETIME DEFAULT CURRENT_TIMESTAMP,
    ip_address     VARCHAR(45),
    FOREIGN KEY (user_id) REFERENCES Users(user_id)
) ENGINE=InnoDB;

-- Table: Notifications
-- Purpose: In-app notification system.
CREATE TABLE Notifications (
    notif_id   INT AUTO_INCREMENT PRIMARY KEY,
    user_id    INT,
    message    TEXT,
    is_read    BOOLEAN DEFAULT FALSE,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    link_to    VARCHAR(255),
    FOREIGN KEY (user_id) REFERENCES Users(user_id)
) ENGINE=InnoDB;

-- Table: Saved_Reports
-- Purpose: Metadata for exported reports.
CREATE TABLE Saved_Reports (
    report_id    INT AUTO_INCREMENT PRIMARY KEY,
    generated_by INT,
    report_type  VARCHAR(50),
    filters_json TEXT,
    generated_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    file_path    VARCHAR(255),
    FOREIGN KEY (generated_by) REFERENCES Users(user_id)
) ENGINE=InnoDB;

Query 1: List All Users With Their Roles and Departments
Retrieves user accounts joined with their assigned role name and department name.
BASIC — JOIN + ORDER BY 
-- Query 1: List all users with their role name and department name
-- Uses: INNER JOIN (2 tables), ORDER BY
SELECT
    u.user_id,
    u.name       AS user_name,
    u.email,
    r.role_name,
    d.dept_name  AS department
FROM Users u
    INNER JOIN Roles r        ON u.role = r.role_id
    INNER JOIN Departments d  ON u.department_id = d.dept_id
ORDER BY u.name ASC;

Query 5: Approved Requests Awaiting PO Generation
Finds purchase requests approved but not yet converted into a purchase order.
BASIC — LEFT JOIN + WHERE NULL 
-- Query 5: Approved requests without a purchase order yet
-- Uses: LEFT JOIN, WHERE ... IS NULL
SELECT pr.pr_id, pr.item_name, pr.quantity, pr.estimated_cost,

FROM Purchase_Requests pr
    INNER JOIN Users u ON pr.employee_id = u.user_id

Query 7: Invoices With Purchase Order and Vendor Details
Joins invoices with their parent PO and billing vendor for accounts payable view.
BASIC — MULTI-JOIN 
-- Query 7: Invoice details with PO reference and vendor
-- Uses: INNER JOIN (3 tables)
SELECT i.invoice_id, i.po_id, v.company_name AS vendor,
       i.amount, i.status AS invoice_status,
       po.status AS po_status, i.due_date

Query 14: End-to-End Procurement Pipeline Report
Traces a purchase request from submission through approval, PO, goods receipt, and invoice — producing a comprehensive lifecycle view using a 6-table join chain.
ADVANCED — 6-TABLE JOIN 
-- Query 14: Full procurement lifecycle report
-- Uses: 6-table INNER/LEFT JOIN chain across entire workflow
SELECT pr.pr_id, u_emp.name AS employee, pr.item_name,
       pr.quantity, pr.estimated_cost,
       pra.decision AS approval_decision,
       u_mgr.name AS approved_by, po.po_id,
       v.company_name AS vendor, po.total_amount AS po_value,
       po.status AS po_status, gr.received_date,
       gr.condition_notes, i.invoice_id,
       i.amount AS invoice_amount, i.status AS invoice_status
FROM Purchase_Requests pr
    INNER JOIN Users u_emp      ON pr.employee_id = u_emp.user_id
    LEFT JOIN PR_Approvals pra  ON pr.pr_id = pra.pr_id
    LEFT JOIN Users u_mgr       ON pra.manager_id = u_mgr.user_id
    LEFT JOIN Purchase_Orders po ON pr.pr_id = po.pr_id
    LEFT JOIN Vendors v          ON po.vendor_id = v.vendor_id
    LEFT JOIN Goods_Receipt gr   ON po.po_id = gr.po_id
    LEFT JOIN Invoices i         ON po.po_id = i.po_id
ORDER BY pr.pr_id LIMIT 10;
